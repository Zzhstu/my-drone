cmake_minimum_required(VERSION 3.0.2)
project(p450_yolo_compare_tracking)
add_compile_options(-std=c++11)
find_package(catkin REQUIRED COMPONENTS roscpp rospy std_msgs sensor_msgs geometry_msgs cv_bridge image_transport prometheus_msgs spirecv_msgs)
catkin_package(CATKIN_DEPENDS roscpp rospy std_msgs sensor_msgs geometry_msgs cv_bridge image_transport prometheus_msgs spirecv_msgs)
include_directories(${catkin_INCLUDE_DIRS})
add_executable(gimbal_target_controller src/gimbal_target_controller.cpp)
add_dependencies(gimbal_target_controller ${catkin_EXPORTED_TARGETS})
target_link_libraries(gimbal_target_controller ${catkin_LIBRARIES})
catkin_install_python(PROGRAMS scripts/yolo_detect_track_node.py scripts/benchmark_logger.py scripts/send_spirecv_click.py scripts/gimbal_manual_teleop.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS gimbal_target_controller RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config docs tools DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
